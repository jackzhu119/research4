class IdentityTests(unittest.TestCase):
    def setUp(self):
        self.registry = {"version": "2026-11-08-confirmed-origin-kind-v1", "total_active_records": 6,
                         "kind_counts": {"枣仁": 3, "酸枣仁": 3}, "workbooks": []}
        self.metadata, self.split, self.confirmation = [], [], []
        self.historical = {"records": []}
        for group, species, origin in (("ZY01", "枣仁", "陕西延安"), ("ZY03", "酸枣仁", "山西")):
            path = f"data/processed/{origin}{species}.xlsx"
            digest = ("a" if group == "ZY01" else "b") * 64
            self.registry["workbooks"].append({"path": path, "sha256": digest, "record_count": 3,
                                                "origin_group_id": group, "species": species, "origin": origin})
            self.confirmation.append({"group_id": group, "new_path": path, "old_path": f"old/{group}.xlsx",
                "uploaded_filename": f"{origin}{species}.xlsx", "origin": origin, "species": species,
                "record_count": "3", "source_sha256": "c" * 64, "output_sha256": digest})
            for index, role in enumerate(("train", "calibration", "prediction"), 1):
                rid = f"{group}-{index:03d}"
                row = {"record_id": rid, "physical_sample_id": rid,
                       "physical_sample_id_basis": "user_declared_independent_20261108",
                       "origin_group_id": group, "species": species, "origin": origin,
                       "source_workbook": path, "excel_column": chr(ord("A") + index),
                       "scan_header": f"样品{index}", "batch_id": "",
                       "dataset_version": self.registry["version"], "sheet": "吸收系数"}
                self.metadata.append(row)
                self.split.append({**row, "role": role})
                self.historical["records"].append({"record_id": rid, "role": "training" if role == "train" else role})

    def validate(self, *, metadata=None, split=None, confirmation=None, excel=None):
        return worker.validate_identity(self.registry, self.metadata if metadata is None else metadata,
            self.split if split is None else split, self.confirmation if confirmation is None else confirmation,
            self.confirmation if excel is None else excel, self.historical)

    def test_confirmed_identity_preserves_old_roles_without_conflating_the_two_species(self):
        result = self.validate()
        self.assertEqual(len(result["by_id"]), 6)
        self.assertEqual(result["by_id"]["ZY01-001"]["species"], "枣仁")
        self.assertEqual(result["by_id"]["ZY03-001"]["species"], "酸枣仁")

    def test_duplicate_record_identity_is_rejected(self):
        with self.assertRaises(ValueError):
            self.validate(metadata=[*self.metadata, self.metadata[0]])

    def test_historical_role_and_confirmed_species_cannot_silently_change(self):
        altered_split = copy.deepcopy(self.split); altered_split[0]["role"] = "prediction"
        with self.assertRaises(ValueError):
            self.validate(split=altered_split)
        altered_metadata = copy.deepcopy(self.metadata); altered_metadata[0]["species"] = "酸枣仁"
        with self.assertRaises(ValueError):
            self.validate(metadata=altered_metadata)

    def test_confirmation_csv_excel_disagreement_is_rejected(self):
        altered = copy.deepcopy(self.confirmation); altered[0]["origin"] = "不同产地"
        with self.assertRaises(ValueError):
            self.validate(excel=altered)


class GroupFoldTests(unittest.TestCase):
    @staticmethod
    def rows():
        rows = []
        for species, groups in (("枣仁", ["ZY01", "ZY02", "ZR01", "ZR02", "ZR03"]),
                                ("酸枣仁", ["ZY03", "ZY04", "ZY05", "ZY06", "ZY07"])):
            for group in groups:
                for index in (1, 2):
                    rows.append({"record_id": f"{group}-{index:03d}", "physical_sample_id": f"{group}-{index:03d}",
                                 "species": species, "origin_group_id": group, "origin": group,
                                 "role": "train"})
        return rows

    def test_paired_folds_hold_out_one_group_per_species_and_cover_each_train_pool_record_once(self):
        rows = self.rows(); by_id = {row["record_id"]: row for row in rows}
        folds, permutations = worker.paired_group_folds(rows, np)
        self.assertEqual(len(folds), 5)
        expected = [{"ZY01", "ZY03"}, {"ZY02", "ZY04"}, {"ZR02", "ZY05"},
                    {"ZR01", "ZY07"}, {"ZR03", "ZY06"}]
        covered = []
        for fold, expected_groups in zip(folds, expected):
            train_ids, test_ids = fold["train_ids"], fold["test_ids"]
            train_groups = {by_id[rid]["origin_group_id"] for rid in train_ids}
            test_groups = {by_id[rid]["origin_group_id"] for rid in test_ids}
            self.assertFalse(set(train_ids) & set(test_ids))
            self.assertFalse(train_groups & test_groups)
            self.assertEqual(test_groups, expected_groups)
            self.assertEqual({by_id[rid]["species"] for rid in test_ids}, {"枣仁", "酸枣仁"})
            covered.extend(test_ids)
        self.assertEqual(sorted(covered), sorted(by_id))
        self.assertEqual(len(covered), len(set(covered)))

    def test_group_control_never_accepts_calibration_or_prediction_role_rows(self):
        rows = self.rows(); rows[0]["role"] = "calibration"
        with self.assertRaises(ValueError):
            worker.paired_group_folds(rows, np)

    def test_twenty_fake_predictors_permute_only_fit_labels_and_leave_truth_and_features_untouched(self):
        rows = self.rows(); original_rows = copy.deepcopy(rows)
        axis = np.linspace(.1, 1.5, 612)
        X = np.vstack([np.sin((index + 1) * axis) + .2 * np.cos((index + 3) * axis) + .01 * index * axis ** 2
                       for index in range(len(rows))])
        original_X = X.copy()
        def snv(matrix):
            return (matrix - matrix.mean(axis=1, keepdims=True)) / matrix.std(axis=1, ddof=1, keepdims=True)
        calls = []
        def fake_predict(train, labels, test):
            calls.append((train.copy(), labels.copy(), test.copy()))
            return np.full(len(test), labels[0])
        protocol = json.loads((AUDIT_ROOT / "preparation/protocol_ROOT_AUTHORIZED.json").read_text())
        with patch.object(worker, "fit_fixed_fullsvm", side_effect=AssertionError("Real fitting is forbidden in tests")):
            summary, predictions, receipts = worker.run_group_control(X, axis, rows, protocol, np,
                                                                     fit_predict=fake_predict, snv=snv)
        self.assertEqual(summary["actual_fits"], 20)
        self.assertEqual(len(calls), 20)
        self.assertEqual(len(receipts), 20)
        np.testing.assert_array_equal(X, original_X)
        self.assertEqual(rows, original_rows)
        lookup = {row["record_id"]: index for index, row in enumerate(rows)}
        for (seen_train, seen_labels, seen_test), receipt in zip(calls, receipts):
            train = np.array([lookup[rid] for rid in receipt["train_ids"]])
            test = np.array([lookup[rid] for rid in receipt["test_ids"]])
            true_fit = np.array([rows[index]["species"] for index in train])
            expected = true_fit.copy()
            seed = receipt["permutation_seed"]
            if seed is not None:
                expected = np.random.default_rng(np.random.SeedSequence([20261108, seed, receipt["fold"]])).permutation(expected)
            np.testing.assert_array_equal(seen_labels, expected)
            np.testing.assert_allclose(seen_train, snv(X[train]), atol=1e-14)
            np.testing.assert_allclose(seen_test, snv(X[test]), atol=1e-14)
            self.assertFalse(receipt["test_truth_permuted"])
            self.assertEqual(receipt["master_roles_accessed_for_spectra"], ["train"])
        for row in predictions:
            self.assertEqual(row["y_true"], rows[lookup[row["record_id"]]]["species"])
        self.assertEqual(len(predictions), 4 * len(rows))
        self.assertTrue(summary["seeds_do_not_increase_independent_N"])
        self.assertIn("NOT_COMPUTED", summary["p_value"])

    def test_failed_fake_fit_has_no_completed_receipt_but_keeps_prior_completed_callbacks(self):
        rows = self.rows(); axis = np.linspace(.1, 1.5, 612)
        X = np.vstack([np.sin((i + 1) * axis) for i in range(len(rows))])
        progress = []; calls = []
        def fake_predict(train, labels, test):
            calls.append(1)
            if len(calls) == 3:
                raise RuntimeError("manufactured third fit failure")
            return np.full(len(test), labels[0])
        protocol = json.loads((AUDIT_ROOT / "preparation/protocol_ROOT_AUTHORIZED.json").read_text())
        with self.assertRaisesRegex(RuntimeError, "third fit failure"):
            worker.run_group_control(X, axis, rows, protocol, np, fit_predict=fake_predict, snv=lambda a: a.copy(),
                                     progress=lambda event, body: progress.append((event, copy.deepcopy(body))))
        self.assertEqual(sum(event == "fit_started" for event, _ in progress), 3)
        completed = [body for event, body in progress if event == "fit_completed"]
        self.assertEqual(len(completed), 2)
        self.assertTrue(all(body["receipt"]["master_roles_accessed_for_spectra"] == ["train"] for body in completed))
        self.assertTrue(all(body["predictions"] for body in completed))

    def test_default_pipeline_passes_only_training_rows_to_fake_scaler_and_fixed_svc(self):
        observed = {}; train = np.ones((3, 612)); test = np.zeros((2, 612)); labels = np.array(["枣仁", "酸枣仁", "酸枣仁"])
        class FakeScaler:
            def fit(self, values):
                observed["scaler_fit"] = values.copy(); return self
            def transform(self, values):
                return values.copy()
        class FakeSVC:
            def __init__(self, **kwargs):
                observed["svc_parameters"] = kwargs
            def fit(self, values, y):
                observed["svc_fit"] = values.copy(); observed["svc_labels"] = y.copy(); return self
            def predict(self, values):
                observed["svc_predict"] = values.copy(); return np.array(["枣仁"] * len(values))
        package, preprocessing, svm = ModuleType("sklearn"), ModuleType("sklearn.preprocessing"), ModuleType("sklearn.svm")
        package.__path__ = []
        preprocessing.StandardScaler, svm.SVC = FakeScaler, FakeSVC
        with patch.dict(sys.modules, {"sklearn": package, "sklearn.preprocessing": preprocessing, "sklearn.svm": svm}):
            predicted = worker.fit_fixed_fullsvm(train, labels, test)
        np.testing.assert_array_equal(observed["scaler_fit"], train)
        np.testing.assert_array_equal(observed["svc_fit"], train)
        np.testing.assert_array_equal(observed["svc_predict"], test)
        self.assertEqual(observed["svc_parameters"], {"C": 10., "gamma": "scale", "class_weight": "balanced",
                         "random_state": 42, "decision_function_shape": "ovr", "break_ties": False})
        self.assertEqual(len(predicted), 2)


class PredictionBoundaryTests(unittest.TestCase):
    def setUp(self):
        self.by_id = {identifier: {"record_id": identifier, "role": "prediction", "species": species,
                                  "origin_group_id": origin}
                      for identifier, species, origin in (("a1", "枣仁", "ZR01"), ("a2", "枣仁", "ZR02"),
                                                          ("b1", "酸枣仁", "ZY03"), ("b2", "酸枣仁", "ZY04"))}

    def test_exact_joint_correctness_is_not_the_product_of_marginal_accuracies(self):
        gate = {"a1": "枣仁", "a2": "枣仁", "b1": "枣仁", "b2": "枣仁"}
        branches = {"枣仁": {"a1": "ZR01", "a2": "ZR02"}, "酸枣仁": {"b1": "WRONG", "b2": "WRONG"}}
        summary, rows = worker.joint_correctness(gate, branches, self.by_id)
        self.assertEqual(summary["N"], 4)
        self.assertEqual(summary["exact_joint_correct_count"], 2)
        self.assertEqual(summary["exact_joint_accuracy"], .5)
        self.assertNotEqual(summary["exact_joint_accuracy"], .5 * .5)
        self.assertEqual(summary["full_joint_macro_f1"], "NOT_COMPUTED")
        wrong = [row for row in rows if not row["species_correct"]]
        self.assertEqual(len(wrong), 2)
        self.assertTrue(all(row["predicted_origin_group"] == "NOT_COMPUTED_WRONG_ROUTE" for row in wrong))
        self.assertTrue(all(row["joint_correct"] == 0 and row["origin_component_evaluated"] == 0 for row in wrong))

    def test_joint_needs_exact_gate_coverage_and_correct_route_origin_predictions(self):
        branches = {"枣仁": {"a1": "ZR01", "a2": "ZR02"}, "酸枣仁": {"b1": "ZY03", "b2": "ZY04"}}
        gate = {key: row["species"] for key, row in self.by_id.items()}
        with self.assertRaises(ValueError):
            worker.joint_correctness({key: value for key, value in gate.items() if key != "a1"}, branches, self.by_id)
        branches["枣仁"].pop("a1")
        with self.assertRaises(ValueError):
            worker.joint_correctness(gate, branches, self.by_id)

    def task_predictions(self):
        return [{"task": "binary_kinds", "model": "full_svm", "method": "source_only", "shots_per_class": "0",
                 "repeat": "0", "record_id": key, "y_true": row["species"], "y_predicted": row["species"],
                 "is_correct": "1"} for key, row in self.by_id.items()]

    def test_fixed_prediction_truth_ids_roles_and_each_model_key_are_audited(self):
        predictions = self.task_predictions()
        summary, selected, diagnostic = worker.audit_task_predictions("binary_kinds", iter(predictions), self.by_id)
        self.assertEqual(summary[0]["correct_count"], 4)
        self.assertEqual(set(selected[("full_svm", "source_only", "0", "0")]), set(self.by_id))
        self.assertEqual(diagnostic["rows_verified"], 4)
        for changed in (predictions + [predictions[0]], predictions[:-1]):
            with self.assertRaises(ValueError):
                worker.audit_task_predictions("binary_kinds", iter(changed), self.by_id)
        changed = copy.deepcopy(predictions); changed[0]["y_true"] = "酸枣仁"
        with self.assertRaises(ValueError):
            worker.audit_task_predictions("binary_kinds", iter(changed), self.by_id)
        wrong_role = copy.deepcopy(self.by_id); wrong_role["a1"]["role"] = "calibration"
        with self.assertRaises(ValueError):
            worker.audit_task_predictions("binary_kinds", iter(predictions), wrong_role)
        changed = copy.deepcopy(predictions); changed[0].update(y_predicted="UNKNOWN", is_correct="0")
        with self.assertRaises(ValueError):
            worker.audit_task_predictions("binary_kinds", iter(changed), self.by_id)

    def test_old_group_predictions_can_be_posthoc_recoded_without_becoming_a_new_species_fit(self):
        by_id = {"ZY01-001": {"species": "枣仁", "origin_group_id": "ZY01"},
                 "ZY02-001": {"species": "枣仁", "origin_group_id": "ZY02"},
                 "ZY03-001": {"species": "酸枣仁", "origin_group_id": "ZY03"}}
        rows = [{"method": "R2", "record_id": "ZY01-001", "y_true": "ZY01", "y_predicted": "ZY02", "is_correct": "0"},
                {"method": "R2", "record_id": "ZY02-001", "y_true": "ZY02", "y_predicted": "ZY01", "is_correct": "0"},
                {"method": "R2", "record_id": "ZY03-001", "y_true": "ZY03", "y_predicted": "ZY03", "is_correct": "1"}]
        result = worker.audit_legacy_predictions(iter(rows), by_id)[0]
        self.assertEqual(result["original_group_metrics"]["correct_count"], 1)
        self.assertEqual(result["posthoc_species_recoded_metrics"]["correct_count"], 3)
        self.assertIn("no new fit", result["interpretation"])


class PinnedSNVTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix="snv-unit-", dir=AUDIT_ROOT / "preparation")
        self.directory = Path(self.temp.name)

    def tearDown(self):
        self.temp.cleanup()

    def test_only_pinned_snv_function_is_loaded_with_ddof_one_and_zero_std_behavior(self):
        source = AUDIT_ROOT / "remote_snapshot" / worker.SOURCE_COMMIT / "src/herb_spectra/spectral_models.py"
        snv = worker.load_pinned_snv(source, np)
        values = np.array([[1., 2., 4.]])

        bad = self.directory / "wrong_snv.py"
        bad.write_text("def _snv(values):\n    return values / values.std(axis=1, ddof=0, keepdims=True)\n")

            worker.load_pinned_snv(bad, np)

class XMLFeatureBoundaryTests(unittest.TestCase):
    MAIN = "http://schemas.openxmlformats.org/spreadsheetml/2006/main"
    REL = "http://schemas.openxmlformats.org/officeDocument/2006/relationships"

    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix="xml-unit-", dir=AUDIT_ROOT / "preparation")
        self.directory = Path(self.temp.name)
        self.registry, self.rows, self.references, self.expected = {"workbooks": []}, [], [], {}
        helper_path = AUDIT_ROOT / "remote_snapshot" / worker.SOURCE_COMMIT / "scripts/prepare_species_revision_features.py"
        self.helper = load(helper_path, "species_audit_pinned_xml_test_helper")

    def tearDown(self):
        self.temp.cleanup()

    def add_workbook(self, name, species="枣仁", axis=None):
        axis = np.linspace(.1, 1.5, 612) if axis is None else np.asarray(axis)
        path = self.directory / (name + ".xlsx")
        workbook = ET.Element(f"{{{self.MAIN}}}workbook")
        sheets = ET.SubElement(workbook, f"{{{self.MAIN}}}sheets")
        ET.SubElement(sheets, f"{{{self.MAIN}}}sheet", name="吸收系数", **{f"{{{self.REL}}}id": "rId1"})
        relationships = '<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships"><Relationship Id="rId1" Target="worksheets/sheet1.xml"/></Relationships>'
        worksheet = ET.Element(f"{{{self.MAIN}}}worksheet")
        ET.SubElement(worksheet, f"{{{self.MAIN}}}dimension", ref="A1:A1")
        data = ET.SubElement(worksheet, f"{{{self.MAIN}}}sheetData")
        def cell(row, coordinate, value, *, text=False):
            node = ET.SubElement(row, f"{{{self.MAIN}}}c", r=coordinate)
            if text:
                node.set("t", "inlineStr")
                inline = ET.SubElement(node, f"{{{self.MAIN}}}is")
                ET.SubElement(inline, f"{{{self.MAIN}}}t").text = str(value)
            else:
                ET.SubElement(node, f"{{{self.MAIN}}}v").text = format(float(value), ".17g")
        header = ET.SubElement(data, f"{{{self.MAIN}}}row", r="1")
        for column, value in zip("ABCD", ("频率", "TRAIN-spectrum", "CAL-spectrum", "PRED-spectrum")):
            cell(header, column + "1", value, text=True)
        values = np.cos(axis) + len(self.references)
        for index, (frequency, value) in enumerate(zip(axis, values), 2):
            row = ET.SubElement(data, f"{{{self.MAIN}}}row", r=str(index))
            cell(row, "A" + str(index), frequency); cell(row, "B" + str(index), value)
            cell(row, "C" + str(index), "FORBIDDEN_CAL_FEATURE", text=True)
            cell(row, "D" + str(index), "FORBIDDEN_PRED_FEATURE", text=True)
        with ZipFile(path, "w") as archive:
            archive.writestr("xl/workbook.xml", ET.tostring(workbook))
            archive.writestr("xl/_rels/workbook.xml.rels", relationships)
            archive.writestr("xl/worksheets/sheet1.xml", ET.tostring(worksheet))
        registry_path = "synthetic/" + name + ".xlsx"
        self.registry["workbooks"].append({"path": registry_path, "sha256": worker.sha(path), "origin_group_id": name,
                                            "species": species, "origin": "private-origin", "record_count": 3})
        self.references.append({"registry_path": registry_path, "source": worker.ref(path)})
        for index, (role, column) in enumerate(zip(("train", "calibration", "prediction"), "BCD")):
            identifier = name + "-" + role
            self.rows.append({"record_id": identifier, "role": role, "species": species, "source_workbook": registry_path,
                              "excel_column": column, "scan_header": ("TRAIN", "CAL", "PRED")[index] + "-spectrum",
                              "origin_group_id": name, "origin": "private-origin", "private_numeric_label": 99999999})
            if role == "train":
                self.expected[identifier] = values.copy()
        return path

    def extract(self):
        return worker.extract_train_alpha(self.registry, self.rows, self.references, self.helper, np)

    def test_only_train_columns_reach_the_pinned_numeric_xml_decoder_and_x_has_no_metadata(self):
        self.add_workbook("g1")
        X, axis, rows, audit = self.extract()
        self.assertEqual(X.shape, (1, 612)); self.assertEqual(axis.shape, (612,))
        np.testing.assert_allclose(X[0], self.expected[rows[0]["record_id"]], rtol=0, atol=0)
        self.assertEqual(rows[0]["private_numeric_label"], 99999999)
        self.assertLess(float(X.max()), 2.)
        self.assertEqual(audit["workbooks"][0]["decoded_column_numbers"], [1, 2])
        self.assertFalse(audit["workbooks"][0]["CAL_or_PRED_spectra_extracted"])

    def test_unapproved_cell_decoder_access_is_blocked_and_guard_restored_on_error(self):
        self.add_workbook("g1")
        original_cell_value = self.helper.cell_value
        def wrong_reader(archive, member, strings, columns):
            self.helper.cell_value(ET.Element("c", r="C2"), strings)
            raise AssertionError("CAL decoder access should have been stopped")
        with patch.object(self.helper, "read_sheet", side_effect=wrong_reader):

class ProofGateTests(ProofBoundaryTests):
    def test_failed_live_proof_prevents_metadata_data_scientific_imports_and_output(self):
        imports = []; original_import = builtins.__import__
        def guarded_import(name, *args, **kwargs):
            if name.split(".")[0] in ("numpy", "sklearn", "openpyxl"):
                imports.append(name); raise AssertionError("Scientific import before accepted proof")
            return original_import(name, *args, **kwargs)
        verify = unittest.mock.Mock(side_effect=ValueError("No live species proof"))
        supervisor = SimpleNamespace(verify_resource_supervision=verify)
        with patch.object(worker, "load_module", return_value=supervisor), \
                patch.object(worker, "nofork_metadata", side_effect=AssertionError("metadata too early")) as metadata, \
                patch.object(worker, "execute_modules", side_effect=AssertionError("data too early")) as execute, \
                patch.object(builtins, "__import__", side_effect=guarded_import):
            with self.assertRaisesRegex(ValueError, "No live species proof"):
                worker.run(self.args)
        metadata.assert_not_called(); execute.assert_not_called()
        self.assertEqual(imports, [])
        self.assertFalse(self.args.output.exists())
        self.assertEqual(verify.call_args.kwargs["expected_manifest_sha256"], self.args.manifest_sha256)
        self.assertEqual(verify.call_args.kwargs["expected_proof_path"], self.args.proof)

    def test_preflight_does_not_read_nested_inputs_or_request_live_proof(self):
        self.args.execute = False
        self.payload["inputs"] = {"unreadable": {"path": "does-not-exist", "sha256": "f" * 64}}
        self.manifest.write_text(json.dumps(self.payload)); self.args.manifest_sha256 = worker.sha(self.manifest)
        with patch.object(worker, "load_module", side_effect=AssertionError("preflight needs no proof")):
            result = worker.run(self.args)
        self.assertEqual(result["status"], "NOT_RUN")
        self.assertEqual(result["new_model_fits"], 0)
        self.assertFalse(self.args.output.exists())

        helper = AUDIT_ROOT / "remote_snapshot" / worker.SOURCE_COMMIT / "scripts/prepare_species_revision_features.py"
        code = """

sys.dont_write_bytecode=True
source,helper=map(Path,sys.argv[1:])
