def load(file, name):
    sp = importlib.util.spec_from_file_location(name, file); m = importlib.util.module_from_spec(sp); sp.loader.exec_module(m); return m

    def tearDown(self): self.temp.cleanup()

    def job(self, name, kind, depends, mode=None, cpu=20):
        j = {"id": name, "kind": kind, "operation": kind, "source": self.m.ref(self.worker),
            "parameters": {} if mode is None else {"mode": mode}, "depends_on": depends, "cpu_seconds": cpu,
            "wall_seconds": 20, "max_attempts": 2, "result_schema": "fixture-yanan-result-v2",
            "success_statuses": ["COMPLETED"]}
        if kind == "summary": j.update(allow_partial_dependencies=True, success_statuses=["SUMMARY_COMPLETE", "PARTIAL_BLOCKS_PRESERVED"])
        return j

    def write_manifest(self):
        self.m.save(self.manifest_path, self.manifest); self.manifest_sha = self.m.sha(self.manifest_path)

    def execute(self):
        self.write_manifest()
        return subprocess.run([self.m.PYTHON, str(self.source), "--manifest", str(self.manifest_path),
            "--manifest-sha256", self.manifest_sha, "--attempt", "fixture-round", "--execute"],
            cwd=self.root, text=True, capture_output=True, timeout=10)

    def snapshot(self): return self.m.read(self.m.ledger_path()), self.m.read(self.m.state_path())

    def test_real_three_job_dynamic_dataset_context_proof_wait4_and_done_skip(self):
        before = self.manifest_path.read_text(); complete = self.execute(); self.assertEqual(complete.returncode, 0, complete.stderr)
        ledger, state = self.snapshot(); self.assertEqual(len(ledger["phases"]), 3)
        self.assertTrue(all(v["status"] == "DONE_SEALED" for v in state["jobs"].values()))
        data = self.m.read(self.m.checked(state["results"]["prepare_data"])); npz_sha = data["dataset"]["npz"]["sha256"]
        self.assertNotIn(npz_sha, before); self.assertEqual(before, self.manifest_path.read_text())
        consumer = self.m.read(self.m.checked(state["results"]["consumer"]))
        self.assertEqual(consumer["dynamic_dataset_sha_verified"], npz_sha)
        context = self.m.read(self.m.checked(state["jobs"]["consumer"]["context"]))
        self.assertEqual(context["dataset_result"], state["results"]["prepare_data"])
        self.assertEqual(context["dependencies"]["prepare_data"], state["results"]["prepare_data"])
        actual = 0.
        for row in ledger["phases"]:
            receipt = self.m.read(self.m.checked(row["wait4_receipt"]))
            cost = receipt["ru_utime_seconds"] + receipt["ru_stime_seconds"]
            self.assertGreater(cost, .03); self.assertEqual(row["actual_cpu_seconds"], cost); actual += cost
        totals = self.m.validate_accounting(ledger, state, self.manifest, self.manifest_sha)
        self.assertAlmostEqual(totals["wait4_child_cpu_seconds"], actual)
        self.assertAlmostEqual(totals["round_budget_charged_cpu_seconds"]-totals["round_actual_cpu_seconds"], 1001)
        self.assertAlmostEqual(ledger["aggregate_budget_charged_cpu_seconds"], 83728.759434+totals["round_budget_charged_cpu_seconds"])
        self.assertEqual(ledger["carry"]["permanent_conservative_debit_cpu_seconds"], 1000)
        self.assertEqual(ledger["carry"]["old_nonmeasured_hold_seconds"], 1)
        args = type("Args", (), {"manifest": str(self.manifest_path), "manifest_sha256": self.manifest_sha})()
        with mock.patch.object(self.m, "run_child", side_effect=AssertionError("DONE cannot re-execute")):
            self.m.run_job(args, self.manifest, ledger, state, self.manifest["jobs"][1])

    def test_entered_fit_and_training_are_not_retried_and_partial_report_survives(self):
        self.manifest["jobs"] = [self.job("prepare_data", "data", []),
            self.job("fit_entry", "classical", ["prepare_data"], "fit-entered"),
            self.job("train_entry", "neural", ["prepare_data"], "train-entered"),
            self.job("summary", "summary", ["prepare_data", "fit_entry", "train_entry"])]
        complete = self.execute(); self.assertEqual(complete.returncode, 0, complete.stderr)
        ledger, state = self.snapshot(); self.assertEqual(len(ledger["phases"]), 4)
        for key in ("fit_entry", "train_entry"):
            rows = [r for r in ledger["phases"] if r["job_id"] == key]
            self.assertEqual(len(rows), 1); self.assertFalse(rows[0]["retry_decision"]["allowed"])
            self.assertNotEqual(state["jobs"][key]["status"], "DONE_SEALED")
        summary = self.m.read(self.m.checked(state["results"]["summary"]))
        self.assertEqual(summary["status"], "PARTIAL_BLOCKS_PRESERVED"); self.assertFalse(summary["winner_selected"])

    def test_explicit_six_zero_prefit_failure_retries_once_with_both_real_costs(self):
        self.manifest["jobs"][1]["parameters"]["mode"] = "pre-fit-error"
        complete = self.execute(); self.assertEqual(complete.returncode, 0, complete.stderr)
        ledger, state = self.snapshot(); rows = [r for r in ledger["phases"] if r["job_id"] == "consumer"]
        self.assertEqual(len(rows), 2); self.assertTrue(rows[0]["retry_decision"]["allowed"])
        self.assertTrue(all(r["actual_cpu_seconds"] > .03 for r in rows)); self.assertEqual(state["jobs"]["consumer"]["status"], "DONE_SEALED")
        self.assertEqual(len(ledger["phases"]), 4)

    def test_budget_not_run_large_job_preserves_independent_cheap_job_and_summary(self):
        self.manifest["jobs"] = [self.job("prepare_data", "data", []),
            self.job("too_large", "classical", ["prepare_data"], cpu=90000),
            self.job("cheap", "classical", ["prepare_data"], cpu=5),
            self.job("summary", "summary", ["prepare_data", "too_large", "cheap"], cpu=5)]
        complete = self.execute(); self.assertEqual(complete.returncode, 0, complete.stderr)
        ledger, state = self.snapshot()
        self.assertEqual(state["jobs"]["too_large"]["status"], "NOT_RUN_RESOURCE_OR_DEADLINE")
        self.assertEqual(state["jobs"]["cheap"]["status"], "DONE_SEALED")
        self.assertEqual([row["job_id"] for row in ledger["phases"]], ["prepare_data", "cheap", "summary"])
        summary = self.m.read(self.m.checked(state["results"]["summary"]))
        self.assertEqual(summary["status"], "PARTIAL_BLOCKS_PRESERVED"); self.assertFalse(summary["winner_selected"])
        self.assertNotIn("too_large", self.m.read(self.m.checked(state["jobs"]["summary"]["context"]))["dependencies"])
