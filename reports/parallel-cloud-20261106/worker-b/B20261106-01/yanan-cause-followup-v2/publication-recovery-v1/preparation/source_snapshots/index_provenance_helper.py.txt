"""Precisely unstage a receipt-bound index; default operation is read-only.

Only ``git restore --staged -- <verified paths>`` can mutate Git state. This
module does not remove locks, reset a branch, clean files, or alter a worktree.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
from datetime import datetime, timezone

SCHEMA = "research4-index-provenance-recovery-v1"


def require(condition, message):
    if not condition:
        raise ValueError(message)


def digest(data):
    return hashlib.sha256(data).hexdigest()


def safe_path(repo, value, *, must_exist=True):
    raw = Path(value)
    require("\x00" not in str(value) and ".." not in raw.parts, "Unsafe path")
    target = raw if raw.is_absolute() else repo / raw
    target = Path(os.path.abspath(target))
    require(target != repo and target.is_relative_to(repo), "Path escapes repository")
    require(not any(part.is_symlink() for part in (target, *target.parents) if part.is_relative_to(repo)), "Symlink path rejected")
    if must_exist:
        require(target.is_file(), "Expected regular file: " + str(target))
    return target


def file_ref(repo, target):
    data = target.read_bytes()
    return {"path": str(target.relative_to(repo)), "bytes": len(data), "sha256": digest(data)}


def run_git(repo, argv):
    env = {key: value for key, value in os.environ.items() if not key.startswith("GIT_")}
    env.update(GIT_OPTIONAL_LOCKS="0", GIT_CONFIG_NOSYSTEM="1")
    result = subprocess.run(["git", "-c", "core.fsmonitor=false", "-C", str(repo), *argv],
                            stdout=subprocess.PIPE, stderr=subprocess.PIPE, env=env, check=False)
    require(result.returncode == 0, "Git failed: " + repr(argv) + ": " + result.stderr.decode("utf-8", "replace")[:2000])
    return result.stdout


def staged_paths(repo, git):
    raw = git(repo, ["diff", "--cached", "--name-only", "--no-renames", "--no-ext-diff", "--no-textconv", "-z", "--"])
    require(not raw or raw.endswith(b"\0"), "Malformed staged path list")
    paths = [item.decode("utf-8") for item in raw.split(b"\0") if item]
    require(len(paths) == len(set(paths)), "Duplicate staged paths")
    return paths


def index_entries(repo, git):
    raw = git(repo, ["ls-files", "--stage", "-z"])
    require(not raw or raw.endswith(b"\0"), "Malformed index entries")
    entries = []
    for item in raw.split(b"\0"):
        if not item:
            continue
        header, name = item.split(b"\t", 1)
        mode, oid, stage = header.decode("ascii").split(" ")
        require(stage == "0", "Unmerged index entry rejected")
        require(re.fullmatch(r"[0-9a-f]{40}|[0-9a-f]{64}", oid), "Invalid blob oid")
        name = name.decode("utf-8")
        safe_path(repo, name, must_exist=False)
        entries.append({"path": name, "mode": mode, "oid": oid, "stage": 0})
    require(len({entry["path"] for entry in entries}) == len(entries), "Duplicate index entry")
    return entries


def read_receipt(repo, receipt, expected_sha, expected_count):
    require(re.fullmatch(r"[0-9a-f]{64}", expected_sha or ""), "Invalid receipt sha256")
    receipt = safe_path(repo, receipt)
    data = receipt.read_bytes()
    require(digest(data) == expected_sha, "Failure receipt hash changed")
    value = json.loads(data)
    artifacts = value.get("artifact_hashes")
    require(isinstance(artifacts, dict) and len(artifacts) == expected_count, "Receipt artifact count differs")
    normalized = {}
    for name, expected in artifacts.items():
        require(isinstance(name, str) and isinstance(expected, dict), "Malformed artifact ref")
        target = safe_path(repo, name)
        relative = str(target.relative_to(repo))
        require(relative not in normalized, "Duplicate normalized artifact path")
        require(type(expected.get("bytes")) is int and expected["bytes"] >= 0 and
                re.fullmatch(r"[0-9a-f]{64}", expected.get("sha256", "")), "Malformed artifact hash/size")
        normalized[relative] = {"bytes": expected["bytes"], "sha256": expected["sha256"]}
    return normalized


def verify_snapshot(repo, receipt, receipt_sha, expected_count, git):
    artifacts = read_receipt(repo, receipt, receipt_sha, expected_count)
    staged = staged_paths(repo, git)
    require(len(staged) == expected_count and set(staged) == set(artifacts), "Staged set does not exactly equal receipt")
    entries = index_entries(repo, git)
    by_path = {entry["path"]: entry for entry in entries}
    blob_data = {}
    blob_refs = []
    worktree = []
    for name in sorted(artifacts):
        require(name in by_path and by_path[name]["mode"] in ("100644", "100755"), "Artifact lacks a regular stage-0 blob")
        entry = by_path[name]
        blob = git(repo, ["cat-file", "blob", entry["oid"]])
        actual = {"bytes": len(blob), "sha256": digest(blob)}
        require(actual == artifacts[name], "Staged blob differs from receipt: " + name)
        current = file_ref(repo, safe_path(repo, name))
        require({key: current[key] for key in ("bytes", "sha256")} == artifacts[name], "Worktree differs from receipt: " + name)
        blob_data[entry["oid"]] = blob
        blob_refs.append({"path": name, "oid": entry["oid"], **actual})
        worktree.append(current)
    raw_index_path = git(repo, ["rev-parse", "--git-path", "index"]).decode("utf-8").strip()
    index = safe_path(repo, raw_index_path)
    index_bytes = index.read_bytes()
    return {"staged_paths": sorted(staged), "index_entries": entries,
            "index": {"path": str(index.relative_to(repo)), "bytes": len(index_bytes), "sha256": digest(index_bytes)}, "staged_blobs": blob_refs,
            "worktree": worktree}, index_bytes, blob_data


def save_json(path, value):
    with path.open("x", encoding="utf-8") as handle:
        json.dump(value, handle, ensure_ascii=False, indent=2, allow_nan=False)
        handle.write("\n")
        handle.flush()
        os.fsync(handle.fileno())


def save_bytes(path, data):
    with path.open("xb") as handle:
        handle.write(data)
        handle.flush()
        os.fsync(handle.fileno())


def perform(repo, receipt, receipt_sha, expected_count, output, *, execute=False, git=run_git):
    repo = Path(os.path.abspath(repo))
    require(repo.is_dir() and not repo.is_symlink(), "Invalid repository")
    require(type(expected_count) is int and expected_count > 0, "Invalid expected count")
    output = safe_path(repo, output, must_exist=False)
    output.mkdir(parents=True, exist_ok=False)
    result = {"schema": SCHEMA, "started_at_utc": datetime.now(timezone.utc).isoformat(),
              "execute": bool(execute), "expected_count": expected_count,
              "failure_receipt": {"path": str(safe_path(repo, receipt).relative_to(repo)), "sha256": receipt_sha},
              "mutation_attempted": False, "status": "REJECTED"}
    try:
        before, index_bytes, blobs = verify_snapshot(repo, receipt, receipt_sha, expected_count, git)
        save_bytes(output / "index_before.bin", index_bytes)
        blob_dir = output / "staged_blobs"
        blob_dir.mkdir()
        for oid, data in blobs.items():
            save_bytes(blob_dir / (oid + ".blob"), data)
        result["before"] = before
        result["index_before_backup"] = file_ref(repo, output / "index_before.bin")
        result["staged_blob_backups"] = [file_ref(repo, p) for p in sorted(blob_dir.iterdir())]
        save_json(output / "before_verification.json", result)
        if not execute:
            result["status"] = "CHECK_ONLY_VERIFIED_NO_MUTATION"
        else:
            # No intervening Git or filesystem writes after this second verification.
            second, second_index, second_blobs = verify_snapshot(repo, receipt, receipt_sha, expected_count, git)
            require(second == before and second_index == index_bytes and second_blobs == blobs,
                    "Index or worktree changed between verification passes")
            result["mutation_attempted"] = True
            git(repo, ["restore", "--staged", "--", *before["staged_paths"]])
            after_staged = staged_paths(repo, git)
            require(after_staged == [], "Staged index is not empty after precise restore")
            after_entries = index_entries(repo, git)
            after_worktree = [file_ref(repo, safe_path(repo, item["path"])) for item in before["worktree"]]
            require(after_worktree == before["worktree"], "Worktree changed during unstage")
            result["after"] = {"staged_paths": after_staged, "staged_index_empty": True,
                               "index_entries": after_entries, "worktree": after_worktree}
            result["status"] = "PRECISE_UNSTAGE_VERIFIED_WORKTREE_UNCHANGED"
    except Exception as exc:
        result.update(error_type=type(exc).__name__, reason=str(exc))
        raise
    finally:
        result["ended_at_utc"] = datetime.now(timezone.utc).isoformat()
        save_json(output / "result.json", result)
    return result


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo", required=True)
    parser.add_argument("--failure-receipt", required=True)
    parser.add_argument("--failure-receipt-sha256", required=True)
    parser.add_argument("--expected-count", type=int, default=92)
    parser.add_argument("--output", required=True, help="New repository-local evidence directory")
    parser.add_argument("--execute", action="store_true")
    args = parser.parse_args(argv)
    try:
        result = perform(args.repo, args.failure_receipt, args.failure_receipt_sha256,
                         args.expected_count, args.output, execute=args.execute)
    except Exception as exc:
        print(json.dumps({"status": "REJECTED", "error_type": type(exc).__name__, "reason": str(exc)}))
        return 2
    print(json.dumps({"status": result["status"], "mutation_attempted": result["mutation_attempted"]}))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
