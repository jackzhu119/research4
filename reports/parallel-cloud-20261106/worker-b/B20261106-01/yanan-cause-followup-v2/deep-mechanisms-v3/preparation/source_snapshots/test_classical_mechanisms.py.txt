class PureAlgebraTests(unittest.TestCase):
    def test_pinned_anchor_has_independent_closed_form_and_exact_binary_replay(self):
        scores = np.array([[-1., 1.], [0., 0.], [1., -1.]])
        labels = np.array([w.CLASSES[1], w.CLASSES[1], w.CLASSES[0]])
        fitted = w.Anchor(w.CLASSES, regularization=1.).fit(scores, labels)
        np.testing.assert_allclose(fitted.coef_, [[.8, .2], [.2, .8]], atol=1e-14)
        np.testing.assert_allclose(fitted.intercept_, [.25, .5], atol=1e-14)
        a, b = w.affine_binary_parameters(fitted.coef_, fitted.intercept_)
        self.assertAlmostEqual(a, 1.2); self.assertAlmostEqual(b, .25)
        np.testing.assert_array_equal(w.threshold_predictions(scores, a, b), fitted.predict(scores))
        np.testing.assert_allclose(w.bias_only(scores, labels), [.25, .5], atol=1e-14)

    def test_positive_reversed_constant_and_exact_tie_use_argmax_first_class(self):
        scores = np.column_stack([-np.array([-2., 0., 1., 2.]), np.array([-2., 0., 1., 2.])])
        cases = [(np.eye(2), np.array([1., -1.])), (-np.eye(2), np.zeros(2)),
                 (np.zeros((2, 2)), np.array([0., 1.])), (np.zeros((2, 2)), np.zeros(2))]
        for coef, bias in cases:
            a, b = w.affine_binary_parameters(coef, bias)
            direct = np.asarray(w.CLASSES)[np.argmax(scores @ coef + bias, axis=1)]
            np.testing.assert_array_equal(w.threshold_predictions(scores, a, b), direct)
        a, b = w.affine_binary_parameters(np.eye(2), [1., -1.])
        self.assertEqual(w.threshold_predictions(np.array([[-1., 1.]]), a, b).tolist(), [w.CLASSES[0]])
        with self.assertRaises(ValueError):
            w.threshold_predictions(np.array([[.1, .3]]), 2., 0.)
        for coef, bias in [(np.eye(3), np.zeros(3)), (np.full((2, 2), np.nan), np.zeros(2))]:
            with self.assertRaises(ValueError): w.affine_binary_parameters(coef, bias)

    def test_normalized_regularization_is_invariant_to_duplicate_objective_terms(self):
        scores = np.array([[-2., 2.], [.3, -.3], [-.1, .1], [1.4, -1.4]])
        labels = np.array([w.CLASSES[1], w.CLASSES[0], w.CLASSES[1], w.CLASSES[0]])
        first = w.Anchor(w.CLASSES, regularization=.4).fit(scores, labels)
        repeated = w.Anchor(w.CLASSES, regularization=1.2).fit(np.tile(scores, (3, 1)), np.tile(labels, 3))
        np.testing.assert_allclose(first.coef_, repeated.coef_, atol=1e-14)
        np.testing.assert_allclose(first.intercept_, repeated.intercept_, atol=1e-14)
        fixed = w.Anchor(w.CLASSES, regularization=.4).fit(np.tile(scores, (3, 1)), np.tile(labels, 3))
        self.assertGreater(np.max(np.abs(first.coef_ - fixed.coef_)), 1e-3)
        for k in (1, 3, 5, 10, 20): self.assertAlmostEqual((k/5)/(2*k), .1)

    def test_balanced_permutation_or_flip_leaves_bias_only_unchanged_and_preserves_truth(self):
        labels = np.array([w.CLASSES[0]]*5+[w.CLASSES[1]]*5); original = labels.copy()
        margins = np.array([-2., -.4, .6, .9, -1.5, .1, 2., 1.1, -.2, .3])
        scores = np.column_stack([-margins, margins])
        for draw in (45, 51, 64):
            perm = w.support_permutation(labels, draw, 5)
            np.testing.assert_array_equal(perm, w.support_permutation(labels, draw, 5))
            self.assertEqual(Counter(perm), Counter(labels))
            np.testing.assert_allclose(w.bias_only(scores, labels), w.bias_only(scores, perm), atol=1e-15)
        flipped = w.flipped_labels(labels)
        np.testing.assert_array_equal(w.flipped_labels(flipped), original)
        np.testing.assert_allclose(w.bias_only(scores, labels), w.bias_only(scores, flipped), atol=1e-15)
        np.testing.assert_array_equal(labels, original)
        with self.assertRaises(ValueError): w.flipped_labels(np.array(['unknown']))

    def test_balanced_anchor_threshold_equals_bias_only_for_every_positive_slope_scale_and_permutation(self):
        margins=np.array([-4.,-2.,-1.,3.,4.,5.]);scores=np.column_stack([-margins,margins])
        labels=np.array([w.CLASSES[0]]*3+[w.CLASSES[1]]*3)
        query=np.linspace(-6,6,1201);q=np.column_stack([-query,query])
        for penalty in (.6,1.,3.):
            for multiplier in (.1,1.,10.):
                for draw in (45,46,47):
                    fit_labels=w.support_permutation(labels,draw,3)
                    model=w.Anchor(w.CLASSES,regularization=penalty).fit(multiplier*scores,fit_labels)
                    unit_a,b=w.affine_binary_parameters(model.coef_,model.intercept_);raw_a=multiplier*unit_a
                    expected_threshold=float(margins.sum()/(len(margins)+penalty))
                    if abs(raw_a)>1e-12:self.assertAlmostEqual(-b/raw_a,expected_threshold,places=10)
                    if raw_a>0:
                        bias=w.bias_only(scores,labels,regularization=penalty)
                        np.testing.assert_array_equal(model.predict(multiplier*q),np.asarray(w.CLASSES)[np.argmax(q+bias,axis=1)])

    def test_support_only_erm_threshold_and_predeclared_tie_rule(self):
        scores = np.array([[-2., 2.], [2., -2.], [-1., 1.], [1., -1.]])
        labels = np.array([w.CLASSES[1], w.CLASSES[0], w.CLASSES[1], w.CLASSES[0]])
        t, metadata = w.support_erm_threshold(scores, labels)
        self.assertEqual(t, 0.)
        self.assertEqual(metadata['support_balanced_error'], 0.)
        self.assertFalse(metadata['query_statistics_or_labels_used'])


    def test_positive_score_units_keep_K0_votes_but_not_fixed_lambda_anchor_mapping(self):
        margins=np.array([-4.,-2.,-1.,3.]);scores=np.column_stack([-margins,margins])
        labels=np.array([w.CLASSES[0],w.CLASSES[0],w.CLASSES[1],w.CLASSES[1]])
        query=np.linspace(-3.,2.,501);query_scores=np.column_stack([-query,query])
        original=w.Anchor(w.CLASSES,regularization=1).fit(scores,labels)
        changed=w.Anchor(w.CLASSES,regularization=1).fit(.1*scores,labels)
        np.testing.assert_array_equal(np.argmax(query_scores,axis=1),np.argmax(.1*query_scores,axis=1))
        a,b=w.affine_binary_parameters(original.coef_,original.intercept_)
        unit_a,unit_b=w.affine_binary_parameters(changed.coef_,changed.intercept_)
        self.assertFalse(np.allclose([a,b],[.1*unit_a,unit_b],atol=1e-10))
        self.assertGreater(a,0.);self.assertGreater(unit_a,0.)
        np.testing.assert_array_equal(original.predict(query_scores),changed.predict(.1*query_scores))
        self.assertAlmostEqual(-b/a,float(margins.sum()/(len(margins)+1)))
        other_labels=np.array([w.CLASSES[0],w.CLASSES[1],w.CLASSES[1],w.CLASSES[1]])
        imbalanced=w.Anchor(w.CLASSES,regularization=1).fit(scores,other_labels)
        imbalanced_scaled=w.Anchor(w.CLASSES,regularization=1).fit(.1*scores,other_labels)
        self.assertGreater(np.count_nonzero(imbalanced.predict(query_scores)!=imbalanced_scaled.predict(.1*query_scores)),0)
        source_sd=float(np.std(margins,ddof=0))
        np.testing.assert_allclose(scores/source_sd,(.1*scores)/(.1*source_sd),atol=1e-15)

    def test_unstratified_equal_prior_control_never_reads_labels_for_selection_or_fit(self):
        data=manufactured_data();_,_,query,pool=w.fixed_scopes(data,'ZY03')
        labels_free={'record_ids':data['record_ids']}
        previous=set()
        for k in w.KS:
            chosen=w.unstratified_support(labels_free,pool,45,k)
            self.assertEqual(len(chosen),2*k);self.assertTrue(previous<=set(chosen));self.assertFalse(set(chosen)&set(query));previous=set(chosen)
        margins=np.array([-1.,.5,2.,3.]);scores=np.column_stack([-margins,margins])
        expected=(2-scores.sum(0))/5
        np.testing.assert_allclose(w.equal_prior_center(scores),expected,atol=1e-15)

class RecorderAndFitTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix='SYNTHETIC-v3-classical-', dir=ROOT/'.runtime')
        self.directory = Path(self.temp.name); self.data = manufactured_data(); self.state = state()
        self.rec = w.Recorder(self.directory, self.state, self.data)
    def tearDown(self): self.temp.cleanup()

    def test_predictions_fsync_precedes_any_truth_score_and_survives_scoring_failure(self):
        query = np.array([0,216]); events=[]
        real_fsync = os.fsync
        def fsync(fd): events.append('fsync'); return real_fsync(fd)
        def fail_score(*args):
            events.append('score'); path=self.directory/'predictions-before-scoring.jsonl.txt'
            self.assertTrue(path.exists()); raw=json.loads(path.read_text().splitlines()[0])
            self.assertNotIn('truth',raw); self.assertTrue(raw['truth_not_joined']); self.assertIn('fsync',events)
            raise RuntimeError('Deliberate scorer failure')
        with patch.object(w.os,'fsync',side_effect=fsync), patch.object(w.LEG,'metrics',side_effect=fail_score):
            with self.assertRaisesRegex(RuntimeError,'Deliberate scorer'):
                self.rec.add('fixed','query',query,[w.CLASSES[0],w.CLASSES[1]])
        self.assertLess(events.index('fsync'),events.index('score')); self.assertFalse(self.rec.observations)

    def test_bad_prediction_identity_or_length_is_rejected_before_output_or_score(self):
        with patch.object(w.LEG,'metrics',side_effect=AssertionError('Scoring malformed predictions')) as score:
            for query,pred in [(np.array([0,216]),[w.CLASSES[0]]),(np.array([0,0]),[w.CLASSES[0]]*2)]:
                with self.assertRaises(ValueError):self.rec.add('bad','q',query,pred)
        score.assert_not_called();self.assertFalse((self.directory/'predictions-before-scoring.jsonl.txt').exists())

    def test_fit_scope_rejection_occurs_before_estimator_and_durable_entry(self):
        class Forbidden:
            def fit(*args): raise AssertionError('Forbidden fit entered')
        values = np.ones((2,3)); labels=np.array(w.CLASSES)
        for idx,allowed in [(np.array([0,0]),np.array([0,1])),(np.array([0,1]),np.array([0]))]:
            with self.assertRaises(ValueError): w.source_fit(self.rec,'bad',idx,values,labels,Forbidden(),kind='fake',allowed=allowed)
        self.assertFalse(self.state['new_fitting_entered']);self.assertEqual(self.state['actual_new_model_fit_calls_started'],0)
        self.data['master_roles'][0]='calibration'
        with self.assertRaises(ValueError):w.source_fit(self.rec,'bad',np.array([0,1]),values,labels,Forbidden(),kind='fake',allowed=np.array([0,1]))
        self.assertFalse(self.state['new_fitting_entered'])

    def test_fit_started_is_persisted_before_estimator_failure_and_not_reset_to_zero(self):
        directory=self.directory
        class Broken:
            def fit(inner,values,labels):
                progress=json.loads((directory/'progress.json').read_text());receipts=json.loads((directory/'fit_receipts.json').read_text())
                if not progress['new_fitting_entered'] or receipts[0]['status']!='FIT_STARTED':raise AssertionError('Entry was not durable')
                raise OSError('Deliberate known failure after fit entered')
        with self.assertRaises(OSError):w.source_fit(self.rec,'broken',np.array([0,1]),np.ones((2,3)),np.array(w.CLASSES),Broken(),kind='fake',allowed=np.array([0,1]))
        self.assertTrue(self.state['new_fitting_entered']);self.assertEqual(self.state['actual_new_model_fit_calls_started'],1)
        self.assertEqual(self.state['actual_new_model_fits'],0)

    def test_K_zero_rules_do_not_fit_and_keep_matched_zero_support(self):
        donor,held,query,pool=w.fixed_scopes(self.data,'ZY03');margins=np.linspace(-2,2,len(held));cache={'ids':self.data['record_ids'][held],'scores':np.column_stack([-margins,margins]),'predictions':np.asarray(w.CLASSES)[(margins>0).astype(int)]}
        with patch.object(w,'KS',(0,)), patch.object(w,'load_source_cache',return_value=({},cache,{'synthetic':True})), \
                patch.object(w,'source_fit',side_effect=AssertionError('K0 cannot fit')):
            w.support_mechanisms(self.rec,{'acid_group':'ZY03','modality':'alpha','draw_seeds':[45]}, {}, {})
        self.assertEqual(len(self.rec.observations),1+len(w.RULES));self.assertFalse(self.state['new_fitting_entered'])

        self.assertTrue(all(p==predictions[0] for p in predictions))

class FakeFactorSVC:
    records=[]
    def __init__(self,C,gamma,class_weight,random_state):self.C,self.gamma=C,gamma;self.classes_=np.asarray(w.CLASSES)
    def fit(self,values,labels):
        self._gamma=1/(values.shape[1]*values.var()) if self.gamma=='scale' and values.var()>0 else 1. if self.gamma=='scale' else float(self.gamma)
        type(self).records.append({'C':self.C,'gamma':self.gamma,'actual_gamma':self._gamma,'x':np.asarray(values).copy(),'y':np.asarray(labels).copy()});return self
    def decision_function(self,values):return np.asarray(values)[:,0]
    def predict(self,values):return self.classes_[(self.decision_function(values)>0).astype(int)]

    def test_missing_live_proof_precedes_all_numeric_or_output_entry(self):

        with patch.object(w,'import_file',return_value=supervisor),patch.object(w,'initialize_numerics',side_effect=AssertionError('Numerics entered too soon')) as initialize:

        payload=json.loads(self.context.read_text());payload['job_sha256']='0'*64;w.save(self.context,payload);self.args.context_sha256=w.sha(self.context);self.args.execute=True
        with patch.object(w,'initialize_numerics',side_effect=AssertionError('Too soon')):
            with self.assertRaises(ValueError):w.run(self.args)
