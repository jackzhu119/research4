def load(file, name):
    sp=importlib.util.spec_from_file_location(name,file);m=importlib.util.module_from_spec(sp);sp.loader.exec_module(m);return m


def bytes_sha(file): return hashlib.sha256(Path(file).read_bytes()).hexdigest()


def simple_ref(file): return {'path':str(file),'sha256':bytes_sha(file)}


def write(file, value):
    Path(file).parent.mkdir(parents=True,exist_ok=True)
    Path(file).write_text(json.dumps(value,indent=2)+'\n')
